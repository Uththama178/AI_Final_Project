/** * Renders the clean quiz editor interface inside the respective chapter preview panel. * Asynchronously fetches the component template and injects dynamic questions. * * @param {Array} backendQuestions - Array of question objects from your NLP model. * @param {HTMLElement} containerTarget - The DOM element where the preview HTML should build (e.g., previewBox). */ async function renderCleanQuizPreview(backendQuestions, containerTarget) { if (!backendQuestions || !Array.isArray(backendQuestions)) { console.error("Invalid array format for quiz questions."); return; } try { // 1. 📄 ඔයා වෙනම හදපු HTML Component එක (Layout එක) Fetch කරගන්නවා const response = await fetch('quiz_preview_component.html'); if (!response.ok) { throw new Error(`Failed to load quiz template component: ${response.statusText}`); } let templateWrapperHTML = await response.text(); // Limit to a maximum of 10 questions to maintain structure const questionsToRender = backendQuestions.slice(0, 10); let questionsListHtml = ""; // 2. 🔁 හැම ප්‍රශ්නයකටම අදාළව Dynamic Editable Cards ටික සාදාගන්නවා questionsToRender.forEach((q, index) => { const questionText = q.Question_Text || q.question_text || ""; const optA = q.Option_A || q.option_a || ""; const optB = q.Option_B || q.option_b || ""; const optC = q.Option_C || q.option_c || ""; const optD = q.Option_D || q.option_d || ""; const correctAnswer = (q.Correct_Answer || q.correct_answer || "A").toUpperCase().trim(); questionsListHtml += `
Question 0${index + 1} Editable Field
`; }); // 3. 🧩 මුලින්ම ප්‍රධාන Layout එක containerTarget (previewBox) එකට ඇතුළත් කරනවා containerTarget.innerHTML = templateWrapperHTML; // 4. 🎯 ඉන්පසු Layout එක ඇතුළේ තියෙන නිවැරදි Placeholder Container එක අල්ලලා ප්‍රශ්න ටික ඒකට Inject කරනවා const dynamicContainer = containerTarget.querySelector("#dynamic-questions-container"); if (dynamicContainer) { dynamicContainer.innerHTML = questionsListHtml; } // 5. 🔗 වැදගත්: ඔයාගේ ප්‍රධාන JS එකේ තියෙන "btn-confirm-chapter-quiz" ක්ලාස් එක ඔයාගේ අලුත් HTML එකේ බටන් එකටත් සම්බන්ධ කරනවා const lockBtn = containerTarget.querySelector(".btn-lock-quiz"); if (lockBtn) { lockBtn.classList.add("btn-confirm-chapter-quiz"); } } catch (error) { console.error("Error building the HTML component inside quiz_preview_render.js:", error); } }
